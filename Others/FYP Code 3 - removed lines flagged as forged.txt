"""
Document Forgery Detection System using Character Segmentation
A Computer Vision approach for detecting tampered regions in financial documents
Eldeena Lim - FYP Project
Supports: PNG, TIFF, JPEG, BMP formats
"""

import cv2
import numpy as np
import matplotlib.pyplot as plt
from collections import defaultdict
import warnings
import os
from PIL import Image
warnings.filterwarnings('ignore')

class DocumentForgeryDetector:
    """
    Character-level forgery detection system that identifies inconsistencies
    in font sizes, alignment, and spacing across document regions.
    """
    
    def __init__(self, image_path, is_genuine=False):
        self.image_path = image_path
        self.is_genuine = is_genuine
        
        if not os.path.exists(image_path):
            raise FileNotFoundError(f"Image file not found: {image_path}")
        
        try:
            pil_image = Image.open(image_path)
            image_array = np.array(pil_image)
            if len(image_array.shape) == 2:
                self.gray = image_array
                self.original_image = cv2.cvtColor(self.gray, cv2.COLOR_GRAY2BGR)
            elif image_array.shape[2] == 3:
                self.original_image = cv2.cvtColor(image_array, cv2.COLOR_RGB2BGR)
                self.gray = cv2.cvtColor(self.original_image, cv2.COLOR_BGR2GRAY)
            elif image_array.shape[2] == 4:
                rgb_array = image_array[:, :, :3]
                self.original_image = cv2.cvtColor(rgb_array, cv2.COLOR_RGB2BGR)
                self.gray = cv2.cvtColor(self.original_image, cv2.COLOR_BGR2GRAY)
            else:
                raise ValueError(f"Unsupported image format with shape: {image_array.shape}")
        except Exception as e:
            print(f"PIL loading failed: {e}")
            self.original_image = cv2.imread(image_path)
            if self.original_image is None:
                raise ValueError(f"Failed to load image: {image_path}")
            self.gray = cv2.cvtColor(self.original_image, cv2.COLOR_BGR2GRAY)
        
        self.height, self.width = self.gray.shape
        self.characters = []
        self.text_regions = []
        self.anomalies = []
        self.baseline_stats = None
        
        print(f"Loaded: {os.path.basename(image_path)} ({self.width}x{self.height})")
        
    def preprocess_image(self):
        print("Preprocessing image...")
        binary = cv2.adaptiveThreshold(
            self.gray, 255, 
            cv2.ADAPTIVE_THRESH_GAUSSIAN_C, 
            cv2.THRESH_BINARY_INV, 11, 2
        )
        denoised = cv2.fastNlMeansDenoising(binary, None, 10, 7, 21)
        kernel = cv2.getStructuringElement(cv2.MORPH_RECT, (2, 2))
        morph = cv2.morphologyEx(denoised, cv2.MORPH_CLOSE, kernel)
        self.binary = morph
        print("Preprocessing complete")
        return morph
    
    def detect_text_regions(self):
        print("Detecting text regions...")
        horizontal_projection = np.sum(self.binary, axis=1)
        lines = []
        in_line = False
        start = 0
        threshold = np.max(horizontal_projection) * 0.1
        
        for i, val in enumerate(horizontal_projection):
            if val > threshold and not in_line:
                start = i
                in_line = True
            elif val <= threshold and in_line:
                if i - start > 5:
                    lines.append((start, i))
                in_line = False
        
        self.text_lines = lines
        print(f"  Detected {len(lines)} text lines")
        return lines

    def _is_horizontal_rule(self, line_roi, char_start, char_end, y_start, y_end):
        h = y_end - y_start
        w = char_end - char_start
        if h > 6 or w < 8 * h:
            return False
        row_sums = np.sum(line_roi > 0, axis=1)
        return np.max(row_sums) > 0.70 * w

    def segment_characters(self):
        print("  Segmenting characters...")
        characters = []
        max_char_width = 0.30 * self.width  # Prevent long lines from being chars

        for line_idx, (y_start, y_end) in enumerate(self.text_lines):
            line_roi = self.binary[y_start:y_end, :]
            
            # Skip lines that are mostly empty (likely horizontal rules)
            line_density = np.sum(line_roi > 0) / (line_roi.size)
            if line_density < 0.05:
                continue

            vertical_projection = np.sum(line_roi, axis=0)
            in_char = False
            char_start = 0
            threshold = np.max(vertical_projection) * 0.1 if np.max(vertical_projection) > 0 else 0

            for x, val in enumerate(vertical_projection):
                if val > threshold and not in_char:
                    char_start = x
                    in_char = True
                elif val <= threshold and in_char:
                    char_width = x - char_start
                    if char_width > 3:
                        # Reject ultra-wide segments (table rules)
                        if char_width > max_char_width:
                            in_char = False
                            continue

                        if self._is_horizontal_rule(line_roi, char_start, x, y_start, y_end):
                            in_char = False
                            continue

                        char_height = y_end - y_start
                        aspect_ratio = char_width / char_height if char_height else 0
                        total = char_width * char_height
                        density = np.sum(line_roi[:, char_start:x] > 0) / total if total else 0

                        characters.append({
                            'x': char_start,
                            'y': y_start,
                            'width': char_width,
                            'height': char_height,
                            'aspect_ratio': aspect_ratio,
                            'density': density,
                            'line_idx': line_idx,
                            'bbox': (char_start, y_start, x, y_end)
                        })
                    in_char = False

        self.characters = characters
        print(f"  Segmented {len(characters)} characters (rules filtered)")
        return characters
    
    def calculate_baseline_statistics(self):
        if not self.characters:
            print("  No characters detected for statistics")
            return None
        
        print("  Calculating baseline statistics...")
        heights = [c['height'] for c in self.characters]
        widths = [c['width'] for c in self.characters]
        aspect_ratios = [c['aspect_ratio'] for c in self.characters]
        densities = [c['density'] for c in self.characters]
        
        line_stats = defaultdict(lambda: {'heights': [], 'widths': []})
        for char in self.characters:
            line_idx = char['line_idx']
            line_stats[line_idx]['heights'].append(char['height'])
            line_stats[line_idx]['widths'].append(char['width'])
        
        stats = {
            'height_mean': np.mean(heights),
            'height_std': np.std(heights),
            'height_median': np.median(heights),
            'width_mean': np.mean(widths),
            'width_std': np.std(widths),
            'aspect_ratio_mean': np.mean(aspect_ratios),
            'aspect_ratio_std': np.std(aspect_ratios),
            'density_mean': np.mean(densities),
            'density_std': np.std(densities),
            'line_stats': dict(line_stats),
            'total_chars': len(self.characters)
        }
        
        self.baseline_stats = stats
        print("  Statistics calculated")
        return stats
    
    def detect_anomalies(self, sensitivity=2.0):
        print(f"  Detecting anomalies (sensitivity={sensitivity})...")
        if not self.baseline_stats:
            self.calculate_baseline_statistics()
        if not self.baseline_stats:
            return []
        
        anomalies = []
        stats = self.baseline_stats
        
        for char in self.characters:
            scores = []
            types = []
            
            h_z = abs(char['height'] - stats['height_mean']) / (stats['height_std'] + 1e-6)
            w_z = abs(char['width'] - stats['width_mean']) / (stats['width_std'] + 1e-6)
            ar_z = abs(char['aspect_ratio'] - stats['aspect_ratio_mean']) / (stats['aspect_ratio_std'] + 1e-6)
            d_z = abs(char['density'] - stats['density_mean']) / (stats['density_std'] + 1e-6)
            
            if h_z > sensitivity: scores.append(h_z); types.append('height')
            if w_z > sensitivity: scores.append(w_z); types.append('width')
            if ar_z > sensitivity: scores.append(ar_z); types.append('aspect_ratio')
            if d_z > sensitivity: scores.append(d_z); types.append('density')
            
            if scores:
                anomalies.append({
                    'char': char,
                    'scores': scores,
                    'types': types,
                    'max_score': max(scores),
                    'severity': 'high' if max(scores) > sensitivity * 1.5 else 'medium'
                })
        
        self.anomalies = sorted(anomalies, key=lambda x: x['max_score'], reverse=True)
        print(f"  Detected {len(anomalies)} anomalous characters")
        return self.anomalies
    
    def cluster_anomalous_regions(self, proximity_threshold=70):  # Tighter clustering
        print("  Clustering anomalous regions...")
        if not self.anomalies:
            print("  No anomalies to cluster")
            return []
        
        anomaly_boxes = [a['char']['bbox'] for a in self.anomalies]
        regions = []
        used = set()
        
        for i, (x1, y1, x2, y2) in enumerate(anomaly_boxes):
            if i in used: continue
            region_boxes = [i]
            used.add(i)
            cx1, cy1, cx2, cy2 = x1, y1, x2, y2
            
            for j, (bx1, by1, bx2, by2) in enumerate(anomaly_boxes):
                if j in used: continue
                c1 = ((x1 + x2) / 2, (y1 + y2) / 2)
                c2 = ((bx1 + bx2) / 2, (by1 + by2) / 2)
                if np.sqrt((c1[0] - c2[0])**2 + (c1[1] - c2[1])**2) < proximity_threshold:
                    region_boxes.append(j)
                    used.add(j)
                    cx1, cy1 = min(cx1, bx1), min(cy1, by1)
                    cx2, cy2 = max(cx2, bx2), max(cy2, by2)
            
            pad = 10
            regions.append({
                'bbox': (max(0, cx1 - pad), max(0, cy1 - pad), min(self.width, cx2 + pad), min(self.height, cy2 + pad)),
                'anomaly_count': len(region_boxes),
                'avg_severity': np.mean([self.anomalies[k]['max_score'] for k in region_boxes])
            })
        
        self.suspicious_regions = sorted(regions, key=lambda x: x['avg_severity'], reverse=True)
        print(f"  Found {len(self.suspicious_regions)} suspicious regions")
        return self.suspicious_regions

    def visualize_results(self, show_characters=True, show_anomalies=True, show_regions=True):
        fig, axes = plt.subplots(2, 2, figsize=(16, 12))
        axes[0, 0].imshow(cv2.cvtColor(self.original_image, cv2.COLOR_BGR2RGB))
        axes[0, 0].set_title('Original Document', fontsize=14, fontweight='bold')
        axes[0, 0].axis('off')
        
        axes[0, 1].imshow(self.binary, cmap='gray')
        axes[0, 1].set_title('Preprocessed Binary Image', fontsize=14, fontweight='bold')
        axes[0, 1].axis('off')
        
        char_vis = self.original_image.copy()
        if show_characters and self.characters:
            for c in self.characters:
                x1, y1, x2, y2 = c['bbox']
                cv2.rectangle(char_vis, (x1, y1), (x2, y2), (0, 255, 0), 1)
        axes[1, 0].imshow(cv2.cvtColor(char_vis, cv2.COLOR_BGR2RGB))
        axes[1, 0].set_title(f'Character Segmentation ({len(self.characters)} chars)', fontsize=14, fontweight='bold')
        axes[1, 0].axis('off')
        
        forgery_vis = self.original_image.copy()
        if show_anomalies and self.anomalies:
            for a in self.anomalies:
                x1, y1, x2, y2 = a['char']['bbox']
                color = (0, 0, 255) if a['severity'] == 'high' else (255, 165, 0)
                cv2.rectangle(forgery_vis, (x1, y1), (x2, y2), color, 2)
        
        if show_regions and hasattr(self, 'suspicious_regions'):
            for r in self.suspicious_regions:
                x1, y1, x2, y2 = [int(v) for v in r['bbox']]
                cv2.rectangle(forgery_vis, (x1, y1), (x2, y2), (255, 0, 0), 3)
                cv2.putText(forgery_vis, f"Forged ({r['avg_severity']:.2f})", (x1, y1 - 10),
                           cv2.FONT_HERSHEY_SIMPLEX, 0.6, (255, 0, 0), 2)
        
        title = f'Forgery Detection Results ({len(getattr(self, "suspicious_regions", []))} regions)'
        axes[1, 1].imshow(cv2.cvtColor(forgery_vis, cv2.COLOR_BGR2RGB))
        axes[1, 1].set_title(title, fontsize=14, fontweight='bold')
        axes[1, 1].axis('off')
        plt.tight_layout()
        return fig

    def generate_report(self):
        report = []
        report.append("=" * 60)
        report.append("DOCUMENT FORGERY DETECTION REPORT")
        report.append("=" * 60)
        report.append(f"Document: {os.path.basename(self.image_path)}")
        report.append(f"Type: {'GENUINE (Baseline)' if self.is_genuine else 'UNDER ANALYSIS'}")
        report.append(f"Total Characters Detected: {len(self.characters)}")
        report.append("")
        
        if self.baseline_stats:
            s = self.baseline_stats
            report.append("CHARACTER STATISTICS:")
            report.append(f"  Height: μ={s['height_mean']:.2f}, σ={s['height_std']:.2f}")
            report.append(f"  Width: μ={s['width_mean']:.2f}, σ={s['width_std']:.2f}")
            report.append(f"  Aspect Ratio: μ={s['aspect_ratio_mean']:.3f}, σ={s['aspect_ratio_std']:.3f}")
            report.append("")
        
        if self.anomalies:
            report.append(f"ANOMALIES DETECTED: {len(self.anomalies)}")
            for i, a in enumerate(self.anomalies[:10], 1):
                c = a['char']
                report.append(f"  {i}. Pos: ({c['x']}, {c['y']}) | Size: {c['width']}x{c['height']}")
                report.append(f"     Types: {', '.join(a['types'])} | Score: {a['max_score']:.2f}")
            report.append("")
        
        if hasattr(self, 'suspicious_regions'):
            report.append(f"SUSPICIOUS REGIONS: {len(self.suspicious_regions)}")
            for i, r in enumerate(self.suspicious_regions, 1):
                x1, y1, x2, y2 = r['bbox']
                report.append(f"  Region {i}: [{x1:.0f},{y1:.0f}]→[{x2:.0f},{y2:.0f}] | Count: {r['anomaly_count']}")
            report.append("")
        
        verdict = "GENUINE" if len(self.anomalies) < 5 else "FORGED"
        confidence = min(100, len(self.anomalies) * 8)
        report.append("=" * 60)
        report.append(f"VERDICT: {verdict}")
        report.append(f"CONFIDENCE: {confidence}%")
        report.append("=" * 60)
        return "\n".join(report)


def ensure_output_folder():
    folder = "PNG_results"
    os.makedirs(folder, exist_ok=True)
    return folder


def compare_documents(genuine_path, tampered_path, sensitivity=2.0):
    print("\n" + "="*60)
    print("DOCUMENT FORGERY DETECTION SYSTEM")
    print("="*60)
    
    out_folder = ensure_output_folder()
    genuine_name = os.path.splitext(os.path.basename(genuine_path))[0]
    tampered_name = os.path.splitext(os.path.basename(tampered_path))[0]
    genuine_png = os.path.join(out_folder, f"{genuine_name}_genuine.PNG")
    tampered_png = os.path.join(out_folder, f"{tampered_name}_forged.PNG")
    
    print("\nProcessing genuine document...")
    genuine_detector = DocumentForgeryDetector(genuine_path, is_genuine=True)
    genuine_detector.preprocess_image()
    genuine_detector.detect_text_regions()
    genuine_detector.segment_characters()
    genuine_detector.calculate_baseline_statistics()
    
    print("\nProcessing suspected document...")
    tampered_detector = DocumentForgeryDetector(tampered_path, is_genuine=False)
    tampered_detector.preprocess_image()
    tampered_detector.detect_text_regions()
    tampered_detector.segment_characters()
    tampered_detector.calculate_baseline_statistics()
    tampered_detector.detect_anomalies(sensitivity=sensitivity)
    tampered_detector.cluster_anomalous_regions(proximity_threshold=70)
    
    print("\n" + genuine_detector.generate_report())
    print("\n" + tampered_detector.generate_report())
    
    print("\nGenerating visualizations...")
    genuine_fig = genuine_detector.visualize_results(show_anomalies=False, show_regions=False)
    genuine_fig.savefig(genuine_png, dpi=150, bbox_inches='tight', format='png')
    plt.close(genuine_fig)
    print(f"Saved: {genuine_png}")
    
    tampered_fig = tampered_detector.visualize_results()
    tampered_fig.savefig(tampered_png, dpi=150, bbox_inches='tight', format='png')
    plt.close(tampered_fig)
    print(f"Saved: {tampered_png}")
    
    plt.show()
    return genuine_detector, tampered_detector
